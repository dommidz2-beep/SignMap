 (function() {
 // ---------- Leaflet Map ----------
 const map = L.map('map').setView([40.7128, -74.0060], 13);

 L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
 attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OSM</a>'
 }).addTo(map);

 // ---------- Add Geocoder (address search) ----------
 const geocoder = L.Control.geocoder({
 defaultMarkGeocode: false,
 position: 'topleft',
 placeholder: 'Search address...',
 errorMessage: 'Address not found'
 })
 .on('markgeocode', function(e) {
 const bbox = e.geocode.bbox;
 const center = e.geocode.center;
 if (bbox) {
 map.fitBounds(bbox);
 } else {
 map.setView(center, 16);
 }
 // Add a temporary marker to show the found location
 L.marker(center, {
 icon: L.divIcon({
 className: 'search-result-marker',
 html: '📍',
 iconSize: [30, 30],
 iconAnchor: [15, 30]
 })
 })
 .addTo(map)
 .bindPopup(`<strong>📍 ${e.geocode.name}</strong>`)
 .openPopup();
 })
 .addTo(map);

 // ---------- Custom address search with button ----------
 const addressInput = document.getElementById('addressInput');
 const searchBtn = document.getElementById('searchBtn');

 function performAddressSearch() {
 const query = addressInput.value.trim();
 if (!query) return;
 geocoder.geocode(query, function(results) {
 if (results && results.length > 0) {
 const result = results[0];
 const bbox = result.bbox;
 const center = result.center;
 if (bbox) {
 map.fitBounds(bbox);
 } else {
 map.setView(center, 16);
 }
 L.marker(center, {
 icon: L.divIcon({
 className: 'search-result-marker',
 html: '📍',
 iconSize: [30, 30],
 iconAnchor: [15, 30]
 })
 })
 .addTo(map)
 .bindPopup(`<strong>📍 ${result.name}</strong>`)
 .openPopup();
 } else {
 alert('Address not found. Please try a different search term.');
 }
 });
 }

 searchBtn.addEventListener('click', performAddressSearch);
 addressInput.addEventListener('keypress', function(e) {
 if (e.key === 'Enter') {
 performAddressSearch();
 }
 });

 // ---------- State ----------
 let signs = [];
 let markers = new Map();
 let nextId = 1;

 // ---------- DOM refs ----------
 const signListEl = document.getElementById('signList');
 const signCountEl = document.getElementById('signCount');
 const clearAllBtn = document.getElementById('clearAllBtn');

 // ---------- Helpers ----------
 function generateLabel(lat, lng) {
 return `Talkie #${nextId}`;
 }

 function renderList() {
 if (signs.length === 0) {
 signListEl.innerHTML = `<div class="empty-state">No signs placed yet.<br />Click the map to add one.</div>`;
 signCountEl.textContent = '0';
 return;
 }

 let html = '';
 signs.forEach(s => {
 const coords = `${s.lat.toFixed(5)}, ${s.lng.toFixed(5)}`;
 html += `
 <div class="sign-item" data-id="${s.id}">
 <div class="sign-info">
 <span class="sign-label">${s.label || 'Talkie Sign'}</span>
 <span class="sign-coords">📍 ${coords}</span>
 </div>
 <div class="sign-actions">
 <button class="icon-btn delete" data-id="${s.id}" title="Remove sign">✕</button>
 </div>
 </div>
 `;
 });
 signListEl.innerHTML = html;
 signCountEl.textContent = signs.length;

 document.querySelectorAll('.sign-item .delete').forEach(btn => {
 btn.addEventListener('click', (e) => {
 e.stopPropagation();
 const id = Number(btn.getAttribute('data-id'));
 deleteSignById(id);
 });
 });
 }

 // ---------- CRUD ----------
 function addSign(lat, lng, label) {
 const id = nextId++;
 const newSign = { id, lat, lng, label: label || generateLabel(lat, lng) };
 signs.push(newSign);
 addMarker(newSign);
 renderList();
 saveToLocal();
 return newSign;
 }

 // ---------- Yard Sign Marker ----------
 function createYardSignIcon(label) {
 const displayLabel = label.length > 12 ? label.slice(0, 10) + '…' : label;
 const html = `
 <div class="yard-sign-marker">
 <div class="sign-board">
 <span class="sign-label-text">${displayLabel}</span>
 <span class="talkie-tag">📡</span>
 </div>
 <div class="sign-stake"></div>
 </div>
 `;
 return L.divIcon({
 className: '',
 html: html,
 iconSize: [60, 48],
 iconAnchor: [30, 48],
 popupAnchor: [0, -44]
 });
 }

 function addMarker(sign) {
 const icon = createYardSignIcon(sign.label || 'Talkie');
 const marker = L.marker([sign.lat, sign.lng], { icon: icon, draggable: false })
 .addTo(map)
 .bindPopup(`
 <div style="display:flex; flex-direction:column; gap:4px; background:#fcf9f5; padding:2px 0;">
 <strong style="color:#1f2e3a;">${sign.label || 'Talkie Sign'}</strong>
 <span style="font-size:0.8rem; color:#3a4f5e;">${sign.lat.toFixed(5)}, ${sign.lng.toFixed(5)}</span>
 <button class="btn btn-danger" style="padding:0.2rem 0.6rem; font-size:0.7rem; border-radius:30px; margin-top:4px; background:#c16b6b; color:white; border:none; cursor:pointer;" data-id="${sign.id}">🗑️ Remove</button>
 </div>
 `);
 markers.set(sign.id, marker);

 marker.on('popupopen', function() {
 const popupNode = document.querySelector('.leaflet-popup-content');
 if (!popupNode) return;
 const delBtn = popupNode.querySelector('.btn-danger');
 if (delBtn) {
 delBtn.addEventListener('click', function(e) {
 const id = Number(this.getAttribute('data-id'));
 deleteSignById(id);
 map.closePopup();
 });
 }
 });
 }

 function deleteSignById(id) {
 const idx = signs.findIndex(s => s.id === id);
 if (idx === -1) return;
 const marker = markers.get(id);
 if (marker) {
 map.removeLayer(marker);
 markers.delete(id);
 }
 signs.splice(idx, 1);
 renderList();
 saveToLocal();
 }

 function clearAllSigns() {
 if (signs.length === 0) return;
 if (!confirm('Remove all Talkie yard signs from map?')) return;
 for (let [id, marker] of markers.entries()) {
 map.removeLayer(marker);
 }
 markers.clear();
 signs = [];
 renderList();
 saveToLocal();
 }

 // ---------- Local Storage ----------
 function saveToLocal() {
 try {
 const data = { signs, nextId };
 localStorage.setItem('talkieSignsData', JSON.stringify(data));
 } catch (e) { /* ignore */ }
 }

 function loadFromLocal() {
 try {
 const raw = localStorage.getItem('talkieSignsData');
 if (!raw) return false;
 const data = JSON.parse(raw);
 if (!data.signs || !Array.isArray(data.signs)) return false;
 signs = data.signs;
 nextId = data.nextId || 1;
 signs.forEach(s => addMarker(s));
 renderList();
 return true;
 } catch (e) {
 return false;
 }
 }

 // ---------- Map click ----------
 map.on('click', function(e) {
 const { lat, lng } = e.latlng;
 addSign(lat, lng);
 });

 // ---------- Event listeners ----------
 clearAllBtn.addEventListener('click', clearAllSigns);

 // ---------- Init ----------
 const loaded = loadFromLocal();
 if (!loaded) {
 const demos = [
 { lat: 40.7128, lng: -74.0060 },
 { lat: 40.7142, lng: -74.0080 },
 { lat: 40.7110, lng: -74.0050 },
 ];
 demos.forEach((pos, i) => {
 addSign(pos.lat, pos.lng, `Talkie #${i+1}`);
 });
 if (demos.length) map.setView([demos[0].lat, demos[0].lng], 15);
 } else {
 if (signs.length > 0) {
 const group = L.featureGroup(Array.from(markers.values()));
 map.fitBounds(group.getBounds().pad(0.1));
 }
 }

 setTimeout(() => map.invalidateSize(), 200);
 })();
